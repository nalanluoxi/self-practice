    public static int[] getMin(char[][] arr, int xn, int yn) {
        int minx = -1;
        int miny = -1;
        int minlen = Integer.MAX_VALUE;
        int minB = Integer.MAX_VALUE;
        for (int i = 0; i < arr.length; i++) {
            for (int j = 0; j < arr[0].length; j++) {
                if (arr[i][j] == '#' || arr[i][j] == '.' || arr[i][j] == '*' || arr[i][j]=='?') {
                    continue;
                }
                int temLen = Math.abs(i - xn) + Math.abs(j - yn);
                if (temLen < minlen && minB > arr[i][j] - '0') {
                    minx = i;
                    miny = j;
                    minB = arr[i][j] - '0';
                }
            }
        }

        return new int[]{minx,miny,minB};
    }
